// Pagination utilities
export function paginateResults<T>(
  items: T[],
  options: PaginationOptions = {}
): PaginationResult<T> {
  const {
    page = 1,
    pageSize = DEFAULT_PAGE_SIZE,
    maxResponseSize = DEFAULT_MAX_RESPONSE_SIZE,
    calculateSize = calculateDefaultSize
  } = options;
  
  // Smart pagination with size limits
  let currentSize = BASE_RESPONSE_SIZE;
  let itemCount = 0;
  
  for (const item of pageItems) {
    const itemSize = calculateSize(item);
    if (currentSize + itemSize > maxResponseSize && itemCount > 0) {

    }
    resultItems.push(item);
    currentSize += itemSize;
    itemCount++;
  }

    items: resultItems,

      page,
      pageSize: itemCount,
      totalPages,
      totalItems: items.length,
      hasNextPage,
      hasPreviousPage
    }
  };
}

### 5. URL-Based Source Identification

**Problem**: Difficult to find script IDs for modification

**Enhanced Resolution**:
```typescript
public async resolveSourceTarget(tabId: string, sourceId: string): Promise<any> {
  const registry = this.getSourceRegistry(tabId);
  
  // First, try direct script ID match
  if (registry.has(sourceId)) {
    return registry.get(sourceId);
  }
  
  // Then, try URL pattern matching (case-insensitive)
  const sourceIdLower = sourceId.toLowerCase();
  const matches: any[] = [];
  
  for (const [scriptId, source] of registry.entries()) {
    if (source.url && source.url.toLowerCase().includes(sourceIdLower)) {
      matches.push({ ...source, scriptId });
    }
  }
  
  // Handle multiple matches with helpful error
  if (matches.length > 1) {
    return null; // Triggers detailed error message
  }
  
  return matches.length === 1 ? matches[0] : null;
}
```

### 6. Reliability Improvements

**Problem**: 92% reliability for console/network capture

**Retry Logic**:
```typescript
export function createRetryWrapper<T>(
  fn: () => Promise<T>,
  options: RetryOptions = {}
): () => Promise<T> {
  const {
    maxRetries = DEFAULT_MAX_RETRIES,
    initialDelay = DEFAULT_INITIAL_DELAY,
    maxDelay = DEFAULT_MAX_DELAY,
    backoffFactor = DEFAULT_BACKOFF_FACTOR
  } = options;
  
  return async () => {
    let lastError: Error | undefined;
    
    for (let attempt = 0; attempt <= maxRetries; attempt++) {
      try {
        return await fn();
      } catch (error) {
        lastError = error as Error;
        
        if (attempt < maxRetries) {
          const delay = Math.min(
            initialDelay * Math.pow(backoffFactor, attempt),
            maxDelay
          );
          await new Promise(resolve => setTimeout(resolve, delay));
        }
      }
    }
    
    throw lastError;
  };
}
```
